    lstHasil.Items.Add(i)

For i As Integer = 10 To 1 Step -1
    lstHasil.Items.Add(i)
Next




' Menghapus hasil sebelumnya.
lstHasil.Items.Clear()

' Jika nilai awal lebih kecil atau sama dengan nilai akhir,
' lakukan perulangan naik.
If nilaiAwal <= nilaiAkhir Then

    For i As Integer = nilaiAwal To nilaiAkhir
        lstHasil.Items.Add(i)
    Next

Else

    ' Jika nilai awal lebih besar, lakukan perulangan turun.
    For i As Integer = nilaiAwal To nilaiAkhir Step -1
        lstHasil.Items.Add(i)
    Next

End If

    End If

    ' Memeriksa apakah nilai akhir berupa angka bulat.
    If Not Integer.TryParse(txtNilaiAkhir.Text, nilaiAkhir) Then
        MessageBox.Show("Masukkan nilai akhir berupa angka!")
        txtNilaiAkhir.Focus()
        Return
    End If

    ' Menghapus hasil sebelumnya.
    lstHasil.Items.Clear()

    ' Menentukan arah perulangan.
    If nilaiAwal <= nilaiAkhir Then

        ' Hitung naik.
        For i As Integer = nilaiAwal To nilaiAkhir
            lstHasil.Items.Add(i)
        Next

    Else

        ' Hitung turun.
        For i As Integer = nilaiAwal To nilaiAkhir Step -1
            lstHasil.Items.Add(i)
        Next

    End If

End Sub

Private Sub txtNilaiAwal_KeyPress(sender As Object, e As KeyPressEventArgs) Handles txtNilaiAwal.KeyPress

    ' Mengizinkan angka dan tombol kontrol seperti Backspace.
    ' Menolak huruf dan simbol lainnya.
    If Not Char.IsControl(e.KeyChar) AndAlso Not Char.IsDigit(e.KeyChar) Then
        e.Handled = True
    End If

End Sub
